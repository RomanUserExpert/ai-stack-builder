#!/usr/bin/env bash
# Fix what ESLint can fix, on staged files only.
files=$(git diff --cached --name-only --diff-filter=ACM | grep -E '\.(ts|tsx)$')
[ -z "$files" ] && exit 0
npx eslint --fix $files && git add $files
